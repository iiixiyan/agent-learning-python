### 数据清洗代码示例

```python
"""
data_cleaning.py - 数据清洗示例
包含：HTML去除、空白清理、长度过滤、去重
"""
import re
import hashlib
from typing import List, Optional


def clean_text(text: str) -> Optional[str]:
    """
    清洗单条文本

    Args:
        text: 原始文本

    Returns:
        Optional[str]: 清洗后的文本，过滤掉的返回 None
    """
    # 1. 去除 HTML 标签
    text = re.sub(r'<[^>]+>', '', text)

    # 2. 去除 URL
    text = re.sub(r'https?://\S+', '', text)

    # 3. 去除多余空白
    text = re.sub(r'\s+', ' ', text).strip()

    # 4. 过滤过短文本
    if len(text) < 50:
        return None

    # 5. 过滤过长文本（可选）
    if len(text) > 100000:
        return None

    return text


def deduplicate(texts: List[str]) -> List[str]:
    """
    文本去重（简单版，实际用 MinHash/SimHash）

    Args:
        texts: 文本列表

    Returns:
        List[str]: 去重后的文本列表
    """
    seen = set()
    unique = []

    for text in texts:
        # 用前200字的 hash 作为去重键（简单版）
        key = hashlib.md5(text[:200].encode()).hexdigest()

        if key not in seen:
            seen.add(key)
            unique.append(text)

    return unique


# 示例
if __name__ == "__main__":
    raw_texts = [
        "<p>这是一段包含HTML的文本，需要清洗。</p>",
        "  多余的   空白   需要   清理  ",
        "太短",  # 会被过滤
        "这是一段重复的文本内容，用于测试去重功能。" * 10,
        "这是一段重复的文本内容，用于测试去重功能。" * 10,  # 重复，会被去掉
    ]

    # 清洗
    cleaned = [clean_text(t) for t in raw_texts]
    cleaned = [t for t in cleaned if t is not None]
    print(f"清洗后: {len(cleaned)} 条")

    # 去重
    unique = deduplicate(cleaned)
    print(f"去重后: {len(unique)} 条")
```

## 第二阶段：预训练（Pre-training）

### 目标

让模型学会语言规律和世界知识。**无监督学习**，预测下一个 token。

### 训练目标

```
输入: "我喜欢吃"
         │
         ▼
    ┌─────────┐
    │  模型    │
    └────┬────┘
         │
         ▼
    预测下一个词: "苹果" (概率最高)
              "香蕉" (概率次之)
              "橘子" (概率第三)
              ...

损失函数: 交叉熵（CrossEntropy）
目标: 让正确的下一个词概率最高
```

### 预训练代码示例

```python
"""
pretrain_demo.py - 预训练步骤简化示例
实际预训练需要分布式训练（DeepSpeed/FSDP），这里只展示核心逻辑
"""
import torch
import torch.nn.functional as F


def pretrain_step(model, batch, optimizer, device="cuda"):
    """
    单步预训练

    Args:
        model: 语言模型
        batch: 批次数据，包含 input_ids 和 labels
        optimizer: 优化器
        device: 设备

    Returns:
        float: 损失值
    """
    input_ids = batch["input_ids"].to(device)  # (batch, seq_len)
    labels = batch["labels"].to(device)        # 与 input_ids 相同，但 shift 一位

    # 前向传播
    outputs = model(input_ids)
    logits = outputs.logits  # (batch, seq_len, vocab_size)

    # 计算损失（预测下一个 token）
    # shift：预测第 i 个位置时，用前 i-1 个位置的信息
    shift_logits = logits[:, :-1, :].contiguous()   # 去掉最后一个位置
    shift_labels = labels[:, 1:].contiguous()        # 去掉第一个位置

    loss = F.cross_entropy(
        shift_logits.view(-1, shift_logits.size(-1)),
        shift_labels.view(-1),
        ignore_index=-100  # 忽略填充位置
    )

    # 反向传播
    optimizer.zero_grad()
    loss.backward()

    # 梯度裁剪（防止梯度爆炸）
    torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)

    # 更新参数
    optimizer.step()

    return loss.item()


# 学习率调度：Warmup + Cosine Decay
def get_lr_scheduler(optimizer, warmup_steps, total_steps):
    """
    Warmup + Cosine Decay 学习率调度

    学习率曲线：
    ↑
    │        ╱╲
    │      ╱    ╲
    │    ╱        ╲
    │  ╱            ╲
    │╱                ╲___
    └──────────────────────→ 训练步数
      warmup        decay
    """
    from torch.optim.lr_scheduler import LambdaLR
    import math

    def lr_lambda(step):
        if step < warmup_steps:
            # Warmup 阶段：线性增加
            return step / max(1, warmup_steps)
        else:
            # Decay 阶段：余弦衰减
            progress = (step - warmup_steps) / max(1, total_steps - warmup_steps)
            return 0.5 * (1.0 + math.cos(math.pi * progress))

    return LambdaLR(optimizer, lr_lambda)
```

### 预训练关键技术

| 技术 | 原理 | 效果 |
|------|------|------|
| **混合精度训练**（FP16/BF16） | 用半精度计算，全精度更新 | 省显存、提速 1.5-2x |
| **梯度累积** | 多步累积梯度后再更新 | 小 batch 模拟大 batch |
| **梯度检查点** | 不保存中间激活，重算 | 用计算换显存（省30-50%） |
| **ZeRO 优化** | 分布式切分参数/梯度/优化器状态 | DeepSpeed，支持超大模型 |
| **学习率调度** | Warmup + Cosine Decay | 稳定训练，提升效果 |
| **数据并行** | 多卡各存一份模型，数据切分 | 线性加速（理想情况） |
| **张量并行** | 模型参数切分到多卡 | 支持单卡放不下的模型 |
| **流水线并行** | 模型层切分到多卡 | 支持超深模型 |

## 第三阶段：监督微调（SFT）

### 目标

让模型学会遵循指令、用指定格式回答。

### 数据格式

```json
[
  {
    "instruction": "写一首关于春天的诗",
    "input": "",
    "output": "春风拂面柳丝长，桃花杏花竞芬芳。\n燕子归来寻旧巢，万物复苏好时光。"
  },
  {
    "instruction": "解释什么是RAG",
    "input": "用简单的语言",
    "output": "RAG（检索增强生成）是一种让大模型先从知识库检索相关资料，再根据资料回答问题的技术。它能减少幻觉，提升回答准确性。"
  }
]
```

### SFT 训练代码

```python
"""
sft_training.py - 监督微调（SFT）训练示例
使用 TRL 库的 SFTTrainer
"""
import torch
from datasets import Dataset
from transformers import AutoModelForCausalLM, AutoTokenizer, TrainingArguments
from trl import SFTTrainer, SFTConfig

# 1. 加载基础模型和分词器
model_name = "gpt2"  # 实际用 Llama/Qwen 等
model = AutoModelForCausalLM.from_pretrained(
    model_name,
    torch_dtype=torch.bfloat16,  # 混合精度
    device_map="auto",
)
tokenizer = AutoTokenizer.from_pretrained(model_name)
tokenizer.pad_token = tokenizer.eos_token

# 2. 准备 SFT 数据
sft_data = [
    {
        "text": "### 指令：写一首关于春天的诗\n### 回答：春风拂面柳丝长，桃花杏花竞芬芳。"
    },
    {
        "text": "### 指令：解释什么是RAG\n### 回答：RAG是检索增强生成技术，先检索再生成。"
    },
    # ... 更多数据
]
dataset = Dataset.from_list(sft_data)

# 3. 配置训练参数
config = SFTConfig(
    output_dir="./sft_model",
    num_train_epochs=3,                    # 训练轮数
    per_device_train_batch_size=4,         # 每卡 batch size
    gradient_accumulation_steps=4,         # 梯度累积
    learning_rate=2e-5,                    # 学习率（比预训练小）
    warmup_ratio=0.1,                      # Warmup 比例
    lr_scheduler_type="cosine",            # 余弦衰减
    max_seq_length=2048,                   # 最大序列长度
    logging_steps=10,                       # 日志间隔
    save_steps=100,                         # 保存间隔
    bf16=True,                              # 混合精度
    gradient_checkpointing=True,            # 梯度检查点（省显存）
)

# 4. 创建 Trainer 并训练
trainer = SFTTrainer(
    model=model,
    args=config,
    train_dataset=dataset,
    processing_class=tokenizer,
)

# 5. 开始训练
trainer.train()

# 6. 保存模型
trainer.save_model("./sft_model_final")
tokenizer.save_pretrained("./sft_model_final")

print("✅ SFT 训练完成！")
```

### SFT 关键要点

| 要点 | 说明 |
|------|------|
| **数据质量 > 数量** | 高质量的 1000 条比低质量的 100 万条效果好 |
| **学习率要小** | 通常 1e-5 ~ 5e-5，比预训练小一个数量级 |
| **训练轮数少** | 通常 1-3 轮，多了会过拟合 |
| **格式要统一** | 所有数据用相同的指令-回答格式 |
| **覆盖要全面** | 覆盖目标场景的各种任务类型 |

## 第四阶段：RLHF / DPO

### 目标

让模型输出更符合人类偏好、更安全、更有用。

### RLHF 三步流程

```
┌─────────────────┐    ┌─────────────────┐    ┌─────────────────┐
│  1. 训练奖励模型  │───▶│  2. PPO 强化学习 │───▶│  3. 最终模型     │
│  (Reward Model)  │    │                 │    │                 │
├─────────────────┤    ├─────────────────┤    ├─────────────────┤
│ 人类对回答排序    │    │ 用奖励模型指导   │    │ 更符合人类偏好   │
│ 训练模型打分      │    │ 模型优化         │    │ 更安全更有用     │
└─────────────────┘    └─────────────────┘    └─────────────────┘
```

### DPO（更简单的替代方案）

DPO（Direct Preference Optimization）直接用偏好数据优化，不需要训练奖励模型，更简单稳定。

```python
"""
dpo_training.py - DPO 训练示例
直接用偏好数据优化，不需要奖励模型
"""
import torch
from datasets import Dataset
from transformers import AutoModelForCausalLM, AutoTokenizer
from trl import DPOTrainer, DPOConfig

# 1. 加载 SFT 模型（作为初始模型和参考模型）
model_name = "./sft_model_final"
model = AutoModelForCausalLM.from_pretrained(
    model_name, torch_dtype=torch.bfloat16, device_map="auto"
)
tokenizer = AutoTokenizer.from_pretrained(model_name)

# 2. 准备偏好数据
# chosen: 人类更喜欢的回答
# rejected: 人类不喜欢的回答
preference_data = [
    {
        "prompt": "### 指令：解释什么是RAG\n### 回答：",
        "chosen": "RAG（检索增强生成）是一种让大模型先从知识库检索相关资料，再根据资料回答问题的技术。它能减少幻觉，提升回答准确性。",
        "rejected": "RAG就是检索增强生成，我也不太清楚具体是什么。",
    },
    # ... 更多偏好数据
]
dataset = Dataset.from_list(preference_data)

# 3. 配置 DPO 训练
dpo_config = DPOConfig(
    output_dir="./dpo_model",
    num_train_epochs=1,              # DPO 通常只训练1轮
    per_device_train_batch_size=2,   # batch size 要小
    learning_rate=5e-7,              # 学习率非常小
    beta=0.1,                        # DPO 温度参数（控制偏离参考模型的程度）
    max_length=2048,
    bf16=True,
    gradient_checkpointing=True,
)

# 4. 创建 DPO Trainer
trainer = DPOTrainer(
    model=model,
    args=dpo_config,
    train_dataset=dataset,
    processing_class=tokenizer,
)

# 5. 训练
trainer.train()
trainer.save_model("./dpo_model_final")

print("✅ DPO 训练完成！")
```

### RLHF vs DPO 对比

| 对比 | RLHF | DPO |
|------|------|-----|
| **复杂度** | 高（需要奖励模型+PPO） | 低（直接优化） |
| **稳定性** | 较低（PPO难调） | 高（稳定） |
| **效果** | 好 | 相当（很多场景更好） |
| **计算量** | 大 | 小 |
| **推荐** | 大厂有资源 | 大多数场景首选 |

## 第五阶段：推理部署

### 推理优化技术

| 技术 | 原理 | 效果 |
|------|------|------|
| **量化（INT8/INT4）** | 降低参数精度 | 显存减半，速度提升 |
| **KV Cache** | 缓存历史 K/V 矩阵 | 避免重复计算，提速 |
| **连续批处理** | 动态合并请求 | 提升吞吐量 2-10x |
| **投机解码** | 小模型猜+大模型验 | 提速 2-3 倍 |
| **PagedAttention** | 分页管理 KV Cache | 显存利用率提升，vLLM 核心 |
| **Flash Attention** | 优化注意力计算 | 显存减少，速度提升 |

### vLLM 部署（生产级推理引擎）

```python
"""
vllm_deploy.py - vLLM 推理部署示例
vLLM 是生产级大模型推理引擎，核心是 PagedAttention
"""
from vllm import LLM, SamplingParams

# 1. 加载模型
llm = LLM(
    model="gpt2",              # 模型名称或路径
    tensor_parallel_size=1,    # 张量并行 GPU 数量
    gpu_memory_utilization=0.9, # GPU 显存利用率
    dtype="bfloat16",          # 数据类型
    max_model_len=4096,        # 最大序列长度
)

# 2. 设置采样参数
sampling_params = SamplingParams(
    temperature=0.7,     # 温度（越高越随机）
    top_p=0.9,           # 核采样
    top_k=50,            # Top-K 采样
    max_tokens=512,      # 最大生成 token 数
    repetition_penalty=1.1, # 重复惩罚
)

# 3. 批量生成
prompts = [
    "写一首关于秋天的诗",
    "解释什么是大模型",
    "用Python写一个快速排序",
]
outputs = llm.generate(prompts, sampling_params)

# 4. 输出结果
for prompt, output in zip(prompts, outputs):
    print(f"\n{'='*60}")
    print(f"Prompt: {prompt}")
    print(f"生成: {output.outputs[0].text}")
    print(f"Token 数: {len(output.outputs[0].token_ids)}")
```

### FastAPI 封装 API 服务

```python
"""
api_server.py - FastAPI 封装大模型 API 服务
"""
from fastapi import FastAPI
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel
from vllm import LLM, SamplingParams
import uvicorn

# 初始化 FastAPI
app = FastAPI(title="大模型 API 服务")
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"])

# 加载模型（启动时加载一次）
llm = LLM(model="gpt2", gpu_memory_utilization=0.9)


# 请求模型
class ChatRequest(BaseModel):
    message: str
    temperature: float = 0.7
    max_tokens: int = 512


class ChatResponse(BaseModel):
    response: str
    token_count: int


# 聊天接口
@app.post("/v1/chat/completions", response_model=ChatResponse)
async def chat(req: ChatRequest):
    """
    聊天接口（兼容 OpenAI 格式）
    """
    sampling_params = SamplingParams(
        temperature=req.temperature,
        max_tokens=req.max_tokens,
        top_p=0.9,
    )

    outputs = llm.generate([req.message], sampling_params)
    text = outputs[0].outputs[0].text
    token_count = len(outputs[0].outputs[0].token_ids)

    return ChatResponse(response=text, token_count=token_count)
```

## 学习要点

1. **大模型训练分四步**：预训练→SFT→RLHF/DPO→部署，每步目标不同，数据量和计算量递减
2. **预训练是无监督学习**，预测下一个 token，学会语言规律和世界知识，计算量极大
3. **SFT 让模型学会遵循指令**，数据质量比数量重要，学习率要小（1e-5~5e-5），训练1-3轮
4. **RLHF/DPO 让模型对齐人类偏好**，更安全更有用；DPO 更简单稳定，大多数场景首选
5. **推理优化是生产关键**，量化、KV Cache、连续批处理、PagedAttention（vLLM）是必备技能
6. **作为应用开发者**，重点理解推理和微调（SFT/DPO），预训练了解即可
7. **数据准备是基础**，数据清洗、去重、分词决定了模型的上限
8. **学习率调度**：Warmup + Cosine Decay 是标准配置，稳定训练提升效果
9. **分布式训练**：数据并行、张量并行、流水线并行、ZeRO，支持超大模型训练
10. **生产部署**：vLLM + FastAPI + 负载均衡，是当前主流的大模型服务架构
