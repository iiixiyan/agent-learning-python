### 1. 用户资料管理

"""
user_profile.py - 用户资料管理
"""
from fastapi import APIRouter, Depends, HTTPException, UploadFile, File
from pydantic import BaseModel, EmailStr
from typing import Optional
from minio import Minio
from sqlalchemy.orm import Session
from datetime import datetime
import uuid
import os

from auth_routes import get_current_user, UserInfo
from models import User, UserPreferences
from database import get_db

router = APIRouter(prefix="/user", tags=["用户资料"])

# MinIO 客户端
minio_client = Minio(
    os.getenv("MINIO_ENDPOINT", "localhost:9000"),
    access_key=os.getenv("MINIO_ACCESS_KEY", "admin"),
    secret_key=os.getenv("MINIO_SECRET_KEY", "password"),
    secure=False,
)


# ============ 数据模型 ============
class UserProfileUpdate(BaseModel):
    nickname: Optional[str] = None
    bio: Optional[str] = None
    email: Optional[EmailStr] = None
    phone: Optional[str] = None


class UserPreferencesUpdate(BaseModel):
    language: Optional[str] = "zh-CN"
    theme: Optional[str] = "light"
    email_notification: Optional[bool] = True
    result_per_page: Optional[int] = 20


# ============ 更新个人资料 ============
@router.put("/profile")
async def update_profile(
    profile: UserProfileUpdate,
    current_user: UserInfo = Depends(get_current_user),
    db: Session = Depends(get_db),
):
    """更新个人资料"""
    update_data = profile.dict(exclude_unset=True)

    # 如果修改邮箱，检查是否已被使用
    if "email" in update_data:
        existing = db.query(User).filter(
            User.email == update_data["email"],
            User.id != current_user.id,
        ).first()
        if existing:
            raise HTTPException(status_code=400, detail="邮箱已被使用")

    db.query(User).filter(User.id == current_user.id).update(update_data)
    db.commit()

    return {"message": "资料更新成功", "profile": update_data}


# ============ 上传头像 ============
@router.post("/avatar")
async def upload_avatar(
    file: UploadFile = File(...),
    current_user: UserInfo = Depends(get_current_user),
    db: Session = Depends(get_db),
):
    """上传头像"""
    # 验证文件类型
    if file.content_type not in ["image/jpeg", "image/png", "image/webp"]:
        raise HTTPException(status_code=400, detail="只支持 JPG/PNG/WebP 格式")

    # 验证文件大小（最大 5MB）
    content = await file.read()
    if len(content) > 5 * 1024 * 1024:
        raise HTTPException(status_code=400, detail="头像大小不能超过 5MB")

    # 保存到 MinIO
    file_ext = file.filename.split(".")[-1]
    object_name = f"avatars/{current_user.id}/{uuid.uuid4()}.{file_ext}"

    from io import BytesIO
    minio_client.put_object(
        "kb-bucket",
        object_name,
        BytesIO(content),
        length=len(content),
        content_type=file.content_type,
    )

    avatar_url = f"{os.getenv('MINIO_PUBLIC_URL', 'http://localhost:9000')}/kb-bucket/{object_name}"

    # 更新用户头像
    db.query(User).filter(User.id == current_user.id).update({"avatar_url": avatar_url})
    db.commit()

    return {"avatar_url": avatar_url}


# ============ 获取用户偏好 ============
@router.get("/preferences")
async def get_preferences(
    current_user: UserInfo = Depends(get_current_user),
    db: Session = Depends(get_db),
):
    """获取用户偏好"""
    prefs = db.query(UserPreferences).filter(
        UserPreferences.user_id == current_user.id
    ).first()

    if not prefs:
        # 返回默认偏好
        return {
            "language": "zh-CN",
            "theme": "light",
            "email_notification": True,
            "result_per_page": 20,
        }

    return {
        "language": prefs.language,
        "theme": prefs.theme,
        "email_notification": prefs.email_notification,
        "result_per_page": prefs.result_per_page,
    }


# ============ 更新用户偏好 ============
@router.put("/preferences")
async def update_preferences(
    prefs: UserPreferencesUpdate,
    current_user: UserInfo = Depends(get_current_user),
    db: Session = Depends(get_db),
):
    """更新用户偏好"""
    existing = db.query(UserPreferences).filter(
        UserPreferences.user_id == current_user.id
    ).first()

    if existing:
        update_data = prefs.dict(exclude_unset=True)
        for key, value in update_data.items():
            setattr(existing, key, value)
    else:
        new_prefs = UserPreferences(
            user_id=current_user.id,
            **prefs.dict(exclude_unset=True),
        )
        db.add(new_prefs)

    db.commit()
    return {"message": "偏好更新成功"}
```

"""
password_manager.py - 密码管理
"""
from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel, field_validator, EmailStr
from sqlalchemy.orm import Session
from datetime import datetime, timedelta
import re
import secrets

from auth_routes import get_current_user, UserInfo
from security import hash_password, verify_password
from jwt_utils import create_access_token, decode_token
from models import User
from database import get_db

router = APIRouter(prefix="/user/password", tags=["密码管理"])


# ============ 数据模型 ============
class ChangePasswordRequest(BaseModel):
    old_password: str
    new_password: str

    @field_validator("new_password")
    def validate_password(cls, v):
        if len(v) < 8:
            raise ValueError("密码至少8位")
        if not re.search(r"[A-Z]", v):
            raise ValueError("密码需包含大写字母")
        if not re.search(r"[a-z]", v):
            raise ValueError("密码需包含小写字母")
        if not re.search(r"\d", v):
            raise ValueError("密码需包含数字")
        return v


class ResetPasswordRequest(BaseModel):
    email: EmailStr


class ConfirmResetRequest(BaseModel):
    token: str
    new_password: str


# ============ 修改密码 ============
@router.put("/change")
async def change_password(
    req: ChangePasswordRequest,
    current_user: UserInfo = Depends(get_current_user),
    db: Session = Depends(get_db),
):
    """修改密码"""
    user = db.query(User).filter(User.id == current_user.id).first()

    # 验证旧密码
    if not verify_password(req.old_password, user.password_hash):
        raise HTTPException(status_code=400, detail="旧密码错误")

    # 不能和旧密码相同
    if verify_password(req.new_password, user.password_hash):
        raise HTTPException(status_code=400, detail="新密码不能与旧密码相同")

    # 更新密码
    user.password_hash = hash_password(req.new_password)
    user.password_changed_at = datetime.now()
    db.commit()

    # 使所有现有 Token 失效（加入黑名单）
    # token_blacklist.add(user.id)

    return {"message": "密码修改成功，请重新登录"}


# ============ 请求重置密码 ============
@router.post("/reset-request")
async def request_password_reset(
    req: ResetPasswordRequest,
    db: Session = Depends(get_db),
):
    """请求重置密码（发送邮件）"""
    user = db.query(User).filter(User.email == req.email).first()

    if user:
        # 生成重置 Token（有效期1小时）
        reset_token = create_access_token(
            data={"sub": str(user.id), "type": "password_reset"},
            expires_delta=timedelta(hours=1),
        )

        # 发送邮件（用 Celery 异步发送）
        # send_reset_email.delay(user.email, reset_token)
        print(f"重置密码链接: https://your-domain.com/reset?token={reset_token}")

    # 无论邮箱是否存在都返回相同消息（防枚举）
    return {"message": "如果邮箱已注册，重置链接已发送"}


# ============ 确认重置密码 ============
@router.post("/reset-confirm")
async def confirm_password_reset(
    req: ConfirmResetRequest,
    db: Session = Depends(get_db),
):
    """确认重置密码"""
    # 验证 Token
    payload = decode_token(req.token)
    if not payload or payload.get("type") != "password_reset":
        raise HTTPException(status_code=400, detail="链接无效或已过期")

    user_id = payload.get("sub")
    user = db.query(User).filter(User.id == int(user_id)).first()

    if not user:
        raise HTTPException(status_code=400, detail="用户不存在")

    # 验证新密码强度
    if len(req.new_password) < 8:
        raise HTTPException(status_code=400, detail="密码至少8位")

    # 更新密码
    user.password_hash = hash_password(req.new_password)
    user.password_changed_at = datetime.now()
    db.commit()

    return {"message": "密码重置成功，请登录"}
```

    os: str
    location: Optional[str] = None
    login_at: datetime
    is_current: bool = False

@router.get("/login-history", response_model=List[LoginRecord])
async def get_login_history(
    current_user: UserInfo = Depends(get_current_user),
    db: Session = Depends(get_db),
):
    """获取登录历史（最近20条）"""
    records = (
        db.query(LoginLog)
        .filter(LoginLog.user_id == current_user.id, LoginLog.status == "success")
        .order_by(LoginLog.created_at.desc())
        .limit(20)
        .all()
    )

    result = []
    for record in records:
        # 解析 User-Agent
        ua = user_agents.parse(record.user_agent) if record.user_agent else None
        result.append(LoginRecord(
            id=record.id,
            ip=record.ip_address,
            device=ua.device.family if ua else "未知",
            browser=ua.browser.family if ua else "未知",
            os=ua.os.family if ua else "未知",
            location=None,  # 可通过 IP 库查询
            login_at=record.created_at,
            is_current=False,  # 可通过 Token 判断

"""
api_keys.py - API Key 管理
"""
from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel
from sqlalchemy.orm import Session
from datetime import datetime, timedelta
from typing import List, Optional
import secrets
import hashlib

from auth_routes import get_current_user, UserInfo
from models import APIKey
from database import get_db

router = APIRouter(prefix="/user/api-keys", tags=["API Key"])


# ============ 数据模型 ============
class APIKeyCreate(BaseModel):
    name: str
    expires_in_days: int = 30


class APIKeyInfo(BaseModel):
    id: int
    name: str
    key_prefix: str  # 只显示前几位，如 sk-abc123...
    created_at: datetime
    expires_at: datetime
    last_used_at: Optional[datetime] = None
    is_active: bool


# ============ 创建 API Key ============
@router.post("")
async def create_api_key(
    req: APIKeyCreate,
    current_user: UserInfo = Depends(get_current_user),
    db: Session = Depends(get_db),
):
    """创建 API Key（仅创建时显示完整 Key）"""
    api_key = "sk-" + secrets.token_hex(32)

    # 加密存储（只存 hash，验证时比对）
    key_hash = hashlib.sha256(api_key.encode()).hexdigest()

    # 计算过期时间
    expires_at = datetime.now() + timedelta(days=req.expires_in_days)

    # 存入数据库
    new_key = APIKey(
        user_id=current_user.id,
        name=req.name,
        key_hash=key_hash,
        key_prefix=api_key[:10] + "...",
        expires_at=expires_at,
        is_active=True,
    )
    db.add(new_key)
    db.commit()
    db.refresh(new_key)

    return {
        "id": new_key.id,
        "api_key": api_key,  # 只返回一次
        "message": "请妥善保存，只显示一次",
        "expires_at": expires_at,
    }


# ============ 列出 API Key ============
@router.get("", response_model=List[APIKeyInfo])
async def list_api_keys(
    current_user: UserInfo = Depends(get_current_user),
    db: Session = Depends(get_db),
):
    """列出 API Key（不显示完整 Key）"""
    keys = (
        db.query(APIKey)
        .filter(APIKey.user_id == current_user.id)
        .order_by(APIKey.created_at.desc())
        .all()
    )

    return [
        APIKeyInfo(
            id=key.id,
            name=key.name,
            key_prefix=key.key_prefix,
            created_at=key.created_at,
            expires_at=key.expires_at,
            last_used_at=key.last_used_at,
            is_active=key.is_active,
        )
        for key in keys
    ]


# ============ 删除 API Key ============
@router.delete("/{key_id}")
async def delete_api_key(
    key_id: int,
    current_user: UserInfo = Depends(get_current_user),

    """删除 API Key"""
    key = db.query(APIKey).filter(

    if not key:
        raise HTTPException(status_code=404, detail="API Key 不存在")

    db.delete(key)

    db: Session = Depends(get_db),

    user_id: int,
    current_user: UserInfo = Depends(require_role("admin")),
    db: Session = Depends(get_db),
):
    """启用/禁用用户"""
    user = db.query(User).filter(User.id == user_id).first()

        raise HTTPException(status_code=400, detail="不能禁用自己")

    db.commit()

"""

from fastapi import APIRouter, Depends

    created_at: datetime


@router.get("", response_model=List[OperationLogRecord])
async def list_operation_logs(
    module: Optional[str] = None,
    action: Optional[str] = None,
    user_id: Optional[int] = None,
    start_date: Optional[str] = None,
    end_date: Optional[str] = None,
    page: int = 1,
    page_size: int = 20,
    current_user: UserInfo = Depends(require_role("admin")),
    db: Session = Depends(get_db),
):
    """查询操作日志（仅管理员）"""
    query = db.query(OperationLog)

    if module:
        query = query.filter(OperationLog.module == module)
    if action:
        query = query.filter(OperationLog.action == action)
    if user_id:
        query = query.filter(OperationLog.user_id == user_id)
    if start_date:
        query = query.filter(OperationLog.created_at >= start_date)
    if end_date:
        query = query.filter(OperationLog.created_at <= end_date)

    total = query.count()
    logs = query.order_by(OperationLog.created_at.desc()).offset((page - 1) * page_size).limit(page_size).all()

    return {
        "total": total,
        "page": page,
        "page_size": page_size,
        "items": [
            OperationLogRecord(
                id=log.id,
                user_id=log.user_id,
                username=log.username,
                action=log.action,
                module=log.module,
                target_type=log.target_type,
                target_id=log.target_id,
                detail=log.detail,
                ip_address=log.ip_address,
                created_at=log.created_at,
            )
            for log in logs
        ],
    }

### 7. 账户注销

```python
"""
account_deletion.py - 账户注销
"""
from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel
from sqlalchemy.orm import Session
from datetime import datetime, timedelta

from auth_routes import get_current_user, UserInfo
from security import verify_password
from models import User
from database import get_db

router = APIRouter(prefix="/user/account", tags=["账户管理"])


class DeleteAccountRequest(BaseModel):
    password: str
    confirm: bool = False


@router.post("/delete")
async def request_account_deletion(
    req: DeleteAccountRequest,
    current_user: UserInfo = Depends(get_current_user),
    db: Session = Depends(get_db),
):
    """请求账户注销（30天冷静期）"""
    if not req.confirm:
        raise HTTPException(status_code=400, detail="请确认注销账户")

    user = db.query(User).filter(User.id == current_user.id).first()

    # 验证密码
    if not verify_password(req.password, user.password_hash):
        raise HTTPException(status_code=400, detail="密码错误")

    # 设置注销时间（30天后执行）
    user.deletion_requested_at = datetime.now()
    user.deletion_scheduled_at = datetime.now() + timedelta(days=30)
    user.status = "pending_deletion"
    db.commit()

    return {
        "message": "账户注销申请已提交，30天冷静期后将永久删除",
        "scheduled_at": user.deletion_scheduled_at,
    }


@router.post("/cancel-delete")
async def cancel_account_deletion(
    current_user: UserInfo = Depends(get_current_user),
    db: Session = Depends(get_db),
):
    """取消账户注销"""
    user = db.query(User).filter(User.id == current_user.id).first()

    if user.status != "pending_deletion":
        raise HTTPException(status_code=400, detail="账户未在注销冷静期")

    user.status = "active"
    user.deletion_requested_at = None
    user.deletion_scheduled_at = None
    db.commit()

    return {"message": "账户注销已取消"}
```

5. **登录历史与设备管理是安全必备**：用户可以查看登录记录、登出其他设备
6. **角色权限管理是管理员功能**：用户管理、角色管理、权限分配、操作日志
7. **操作日志是审计合规要求**：记录谁在什么时候做了什么，支持查询和追溯
8. **账户注销要有冷静期**：30天冷静期，期间可以取消，保护用户误操作
9. **用户偏好设置提升体验**：语言、主题、通知、每页结果数，个性化体验
